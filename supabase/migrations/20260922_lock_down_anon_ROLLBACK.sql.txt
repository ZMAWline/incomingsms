-- anon_readonly_all (SELECT, USING true) existed on every table EXCEPT these 13.
DO $$
DECLARE
  t text;
  excluded text[] := ARRAY[
    'dashboard_invites', 'dashboard_users', 'gateways', 'otp_portal_assignments',
    'reseller_api_keys', 'resellers', 'shop_customers', 'shop_sessions',
    'address_pool', 'address_pool_deletions', 'dashboard_api_keys',
    'dashboard_audit_log', 'dashboard_saved_filters'
  ];
BEGIN
  FOR t IN
    SELECT c.relname FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace
    WHERE n.nspname = 'public' AND c.relkind = 'r' AND NOT (c.relname = ANY(excluded))
  LOOP
    EXECUTE format('DROP POLICY IF EXISTS anon_readonly_all ON public.%I', t);
    EXECUTE format('CREATE POLICY anon_readonly_all ON public.%I FOR SELECT TO anon USING (true)', t);
  END LOOP;
END $$;

-- Both roles had USAGE, SELECT, UPDATE on every sequence.
GRANT USAGE, SELECT, UPDATE ON ALL SEQUENCES IN SCHEMA public TO anon, authenticated;
